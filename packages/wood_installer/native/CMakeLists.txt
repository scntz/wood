cmake_minimum_required(VERSION 3.25)
project(wood_installer_native VERSION 0.1.0 LANGUAGES C)
include(CTest)
if(WIN32 AND CMAKE_C_COMPILER_ID MATCHES "Clang")
    if(NOT CMAKE_AR OR CMAKE_AR MATCHES "-NOTFOUND$")
        find_program(setup_archive NAMES llvm-ar llvm-lib lib REQUIRED)
        set(CMAKE_AR "${setup_archive}")
    endif()
    get_filename_component(archive_name "${CMAKE_AR}" NAME_WE)
    if(archive_name STREQUAL "lib" OR archive_name STREQUAL "llvm-lib")
        set(CMAKE_C_ARCHIVE_CREATE "<CMAKE_AR> /nologo /out:<TARGET> <OBJECTS>")
        set(CMAKE_C_ARCHIVE_APPEND "<CMAKE_AR> /nologo /out:<TARGET> <TARGET> <OBJECTS>")
        set(CMAKE_C_ARCHIVE_FINISH "")
    endif()
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$")
    message(FATAL_ERROR "Installer adapters currently support only Windows/Linux x86-64")
endif()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Installer platform is not implemented")
endif()
add_library(wood_installer_native STATIC files.c)
set_target_properties(wood_installer_native PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON MSVC_RUNTIME_LIBRARY "")
if(MSVC AND NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    target_compile_options(wood_installer_native PRIVATE /W4 /WX /GS- /Zl)
else()
    target_compile_options(wood_installer_native PRIVATE -ffreestanding -fno-builtin -fno-stack-protector -Wall -Wextra -Werror)
endif()
option(SETUP_SANITIZERS "Instrument the private adapter and hosted C test only" OFF)
if(BUILD_TESTING)
    add_executable(installer_adapter_tests ../tests/adapter.c)
    target_link_libraries(installer_adapter_tests PRIVATE wood_installer_native)
    set_target_properties(installer_adapter_tests PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/native-tests")
    add_test(NAME installer_adapter COMMAND installer_adapter_tests "${CMAKE_CURRENT_BINARY_DIR}/native-tests")
    if(SETUP_SANITIZERS)
        if(WIN32 OR NOT CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang)$")
            message(FATAL_ERROR "This hosted adapter sanitizer gate is qualified only with Linux GCC/Clang")
        endif()
        target_compile_options(wood_installer_native PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_compile_options(installer_adapter_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(installer_adapter_tests PRIVATE -fsanitize=address,undefined)
    endif()
endif()
if(BUILD_TESTING AND DEFINED WOOD)
    add_test(NAME installer_contracts COMMAND "${CMAKE_COMMAND}"
        "-DWOOD=${WOOD}" "-DADAPTER=$<TARGET_FILE:wood_installer_native>"
        "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/../../.."
        "-DWORK=${CMAKE_CURRENT_BINARY_DIR}/qualification"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/../tests/qualification.cmake")
    set_tests_properties(installer_contracts PROPERTIES TIMEOUT 600)
endif()
