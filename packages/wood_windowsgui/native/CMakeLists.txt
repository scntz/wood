cmake_minimum_required(VERSION 3.25)
project(wood_windowsgui VERSION 0.1.0 LANGUAGES C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$")
    message(FATAL_ERROR "wood_windowsgui requires Windows x86-64; other GUI backends are not implemented")
endif()
add_library(wood_windowsgui STATIC windows.c)
set_target_properties(wood_windowsgui PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON
    MSVC_RUNTIME_LIBRARY "")
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    get_filename_component(llvm_tools "${CMAKE_C_COMPILER}" DIRECTORY)
    if(NOT CMAKE_AR OR CMAKE_AR MATCHES "-NOTFOUND$")
        find_program(wood_archive NAMES llvm-ar llvm-lib lib HINTS "${llvm_tools}" REQUIRED)
        set(CMAKE_AR "${wood_archive}")
    endif()
    get_filename_component(archive_name "${CMAKE_AR}" NAME_WE)
    if(archive_name STREQUAL "lib" OR archive_name STREQUAL "llvm-lib")
        set(CMAKE_C_ARCHIVE_CREATE "<CMAKE_AR> /nologo /out:<TARGET> <OBJECTS>")
        set(CMAKE_C_ARCHIVE_APPEND "<CMAKE_AR> /nologo /out:<TARGET> <TARGET> <OBJECTS>")
        set(CMAKE_C_ARCHIVE_FINISH "")
    endif()
    target_compile_options(wood_windowsgui PRIVATE -ffreestanding -fno-builtin -fno-stack-protector -Wall -Wextra -Werror)
elseif(MSVC)
    target_compile_options(wood_windowsgui PRIVATE /W4 /WX /GS- /Zl)
else()
    message(FATAL_ERROR "Use a Windows Clang or MSVC C17 compiler and matching Windows SDK")
endif()
# Only OS import libraries, explicitly passed by the Wood consumer. The adapter
# itself has no CRT entry points, malloc, C++ objects or retained Wood callbacks.
target_link_libraries(wood_windowsgui INTERFACE user32 gdi32 kernel32)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-inputs-$<CONFIG>.json" CONTENT
    "{\"schema_version\":1,\"package\":\"wood_windowsgui\",\"version\":\"0.1.0\",\"platform\":\"windows-x86_64\",\"abi\":4,\"archive\":\"$<TARGET_FILE:wood_windowsgui>\",\"system_import_libraries\":[\"user32.lib\",\"gdi32.lib\",\"kernel32.lib\"]}\n")
