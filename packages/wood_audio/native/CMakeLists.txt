cmake_minimum_required(VERSION 3.25)
project(wood_audio VERSION 0.1.0 LANGUAGES C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The wood_audio device adapter requires Windows x86-64; no other device backend is implemented")
endif()
add_library(wood_audio STATIC windows.c)
set_target_properties(wood_audio PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON MSVC_RUNTIME_LIBRARY "")
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    target_compile_options(wood_audio PRIVATE -ffreestanding -fno-builtin -fno-stack-protector -Wall -Wextra -Werror)
elseif(MSVC)
    target_compile_options(wood_audio PRIVATE /W4 /WX /GS- /Zl)
else()
    message(FATAL_ERROR "Use a Windows Clang or MSVC C17 compiler")
endif()
