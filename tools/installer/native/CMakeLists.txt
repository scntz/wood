cmake_minimum_required(VERSION 3.25)
project(wood_setup_windows LANGUAGES C)
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    if(NOT CMAKE_AR OR CMAKE_AR MATCHES "-NOTFOUND$")
        find_program(setup_archive NAMES llvm-ar llvm-lib lib REQUIRED)
        set(CMAKE_AR "${setup_archive}")
    endif()
    get_filename_component(archive_name "${CMAKE_AR}" NAME_WE)
    if(archive_name STREQUAL "lib" OR archive_name STREQUAL "llvm-lib")
        set(CMAKE_C_ARCHIVE_CREATE "<CMAKE_AR> /nologo /out:<TARGET> <OBJECTS>")
        set(CMAKE_C_ARCHIVE_APPEND "<CMAKE_AR> /nologo /out:<TARGET> <TARGET> <OBJECTS>")
        set(CMAKE_C_ARCHIVE_FINISH "")
    endif()
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Windows setup frontend requires Windows x86-64")
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../packages/wood_installer/native" engine)
add_library(wood_setup_windows STATIC windows.c integration.c)
add_library(wood_setup_launcher STATIC launcher.c)
set_target_properties(wood_setup_windows PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON MSVC_RUNTIME_LIBRARY "")
set_target_properties(wood_setup_launcher PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON MSVC_RUNTIME_LIBRARY "")
if(MSVC AND NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    target_compile_options(wood_setup_windows PRIVATE /W4 /WX /GS- /Zl)
    target_compile_options(wood_setup_launcher PRIVATE /W4 /WX /GS- /Zl)
else()
    target_compile_options(wood_setup_windows PRIVATE -ffreestanding -fno-builtin -fno-stack-protector -Wall -Wextra -Werror)
    target_compile_options(wood_setup_launcher PRIVATE -ffreestanding -fno-builtin -fno-stack-protector -Wall -Wextra -Werror)
endif()
# Import libraries are build inputs only; no SDK archive is redistributed here.
target_link_libraries(wood_setup_windows INTERFACE user32 gdi32 kernel32 advapi32 comctl32 ole32 shell32 uuid)
